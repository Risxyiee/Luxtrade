    29→// Generate realistic equity curve data
    30→function generateEquityData(points: number): EquityPoint[] {
    31→  const data: EquityPoint[] = []
    32→  let equity = 10500 // Starting equity
    33→  
    34→  for (let i = 0; i < points; i++) {
    35→    // Random walk with slight upward bias
    36→    const change = (Math.random() - 0.45) * 150
    37→    equity = Math.max(9500, Math.min(12000, equity + change))
    38→    data.push({
    39→      time: i,
    40→      value: equity,
    41→      change: change
    42→    })
    43→  }
    44→  return data
    45→}
