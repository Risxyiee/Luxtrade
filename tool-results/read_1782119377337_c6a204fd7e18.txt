   126→   126→  amount,
   127→   127→  expiresAt,

   179→   179→          clearInterval(pollingRef.current)
   180→   180→          pollingRef.current = null
   181→   181→        }

   183→   183→        setOrderStatus('EXPIRED')
   184→   184→        if (pollingRef.current) {
   185→   185→          clearInterval(pollingRef.current)
   186→   186→          pollingRef.current = null
   187→   187→        }
   188→   188→      }
   189→   189→    } catch {
   190→   190→      // Silent fail on polling
   191→   191→    }
   192→   192→  }, [realInvoiceNumber])

   198→   198→      pollingRef.current = setInterval(() => {
   199→   199→        setPollCount(prev => prev + 1)
   200→   200→      }, 5000)
   201→   201→    }
   202→   202→

   204→   204→      if (pollingRef.current) {
   205→   205→        clearInterval(pollingRef.current)
   206→   206→        pollingRef.current = null
   207→   207→      }
   208→   208→    }

   266→   266→        year: 'numeric',

   304→   304→  // MAIN PAY HANDLER
   305→   305→  // ============================================

   308→   308→    setPaying(true)
   309→   309→    setPayError('')

   325→   325→          durationMonths,
   326→   326→          paymentMethod: selectedMethod,
   327→   327→        })

   329→   329→
   330→   330→      const data = await response.json()
   331→   331→
   332→   332→      if (!response.ok || !data.paymentUrl) {
