   120→export default function PaymentInvoiceModal({
   121→  isOpen,
   122→  onClose,
   123→  invoiceNumber,
   124→  planName,
   125→  duration,
   126→  amount,
   127→  expiresAt,
   128→  plan,
   129→  durationMonths,
   130→  paymentUrl: initialPaymentUrl,
   131→  orderId: initialOrderId,
   132→  onPayNow,
   133→  initialStatus,
   134→  paidAt: initialPaidAt,
   135→}: PaymentInvoiceModalProps) {
   136→  const [copied, setCopied] = useState<string | null>(null)
   137→  const [paying, setPaying] = useState(false)
   138→  const [paid, setPaid] = useState(false)
   139→  const [paymentUrl, setPaymentUrl] = useState(initialPaymentUrl || '')
   140→  const [orderId, setOrderId] = useState(initialOrderId || '')

   142→  const [realInvoiceNumber, setRealInvoiceNumber] = useState(invoiceNumber)
   143→
   144→  // Status from DB (polling)
   145→  const [orderStatus, setOrderStatus] = useState<OrderStatus>(initialStatus || 'PENDING')

   156→

   159→  const isExpired = orderStatus === 'EXPIRED'
   160→  const canPay = selectedMethod !== null && !paying && !paid && !isPaid

   163→  // POLLING: Check order status every 5 seconds after payment
   164→  // ============================================
   165→  const pollOrderStatus = useCallback(async () => {
   166→    if (isPaidRef.current || !realInvoiceNumber) return
   167→
   168→    try {
   169→      const res = await fetch(`/api/payment/order-status?invoiceNumber=${encodeURIComponent(realInvoiceNumber)}`)
   170→      if (!res.ok) return
   171→
   172→      const data = await res.json()

   174→        isPaidRef.current = true
   175→        setOrderStatus('SUCCESS')
   176→        setPaidAtDate(data.paidAt || new Date().toISOString())
   177→        // Stop polling on success
   178→        if (pollingRef.current) {
   179→          clearInterval(pollingRef.current)
   180→          pollingRef.current = null
   181→        }
   182→      } else if (data.status === 'EXPIRED') {
   183→        setOrderStatus('EXPIRED')
   184→        if (pollingRef.current) {
   185→          clearInterval(pollingRef.current)
   186→          pollingRef.current = null
   187→        }
   188→      }
   189→    } catch {
   190→      // Silent fail on polling
   191→    }
   192→  }, [realInvoiceNumber])
   193→
   194→  // Start polling when user has paid (opened gateway)
   195→  useEffect(() => {
   196→    if (paid && !isPaidRef.current) {
   197→      // Poll every 5 seconds
   198→      pollingRef.current = setInterval(() => {
   199→        setPollCount(prev => prev + 1)
   200→      }, 5000)
   201→    }
   202→
   203→    return () => {
   204→      if (pollingRef.current) {
   205→        clearInterval(pollingRef.current)
   206→        pollingRef.current = null
   207→      }
   208→    }
   209→  }, [paid])
   210→
   211→  // Actual polling triggered by pollCount changes
   212→  useEffect(() => {
   213→    if (paid && !isPaidRef.current && pollCount > 0) {
   214→      pollOrderStatus()
   215→    }
   216→  }, [paid, pollCount, pollOrderStatus])
   217→

   251→        year: 'numeric',
   252→        hour: '2-digit',
   253→        minute: '2-digit'
   254→      })
   255→    } catch {
   256→      return dateStr
   257→    }
   258→  }

   264→        day: 'numeric',
   265→        month: 'long',
   266→        year: 'numeric',

   320→        method: 'POST',
   321→        headers: { 'Content-Type': 'application/json' },
   322→        body: JSON.stringify({
   323→          amount,
   324→          plan,
   325→          durationMonths,
   326→          paymentMethod: selectedMethod,
   327→        })
   328→      })
   329→
   330→      const data = await response.json()
   331→
   332→      if (!response.ok || !data.paymentUrl) {
   333→        throw new Error(data.error || data.details || 'Gagal membuat pesanan')
   334→      }
   335→
   336→      setPaymentUrl(data.paymentUrl)
   337→      setOrderId(data.orderId || '')
   338→      // Store the real invoice number from API for polling
   339→      if (data.invoiceNumber) {
   340→        setRealInvoiceNumber(data.invoiceNumber)
   341→      }
   342→
   343→      // Open SakuraPay gateway
   344→      window.open(data.paymentUrl, '_blank')
   345→      setPaying(false)
   346→      setPaid(true)
   347→    } catch (error: any) {

   349→      setPaying(false)

   379→        >
   380→          <div className={`

   415→              <div className="relative p-6">
   416→                <div className="flex items-center justify-between">
   417→                  <div className="flex items-center gap-3">
