    97→interface PaymentInvoiceModalProps {
    98→  isOpen: boolean
    99→  onClose: () => void
   100→  invoiceNumber: string
   101→  planName: string
   102→  duration: string
   103→  amount: number
   104→  expiresAt: string
   105→  plan: string        // PRO | LIFETIME
   106→  durationMonths: number
   107→  // paymentUrl is optional now — modal will create its own order
   108→  paymentUrl?: string

   112→
   113→export default function PaymentInvoiceModal({
   114→  isOpen,
   115→  onClose,
   116→  invoiceNumber,
   117→  planName,
   118→  duration,
   119→  amount,
   120→  expiresAt,
   121→  plan,
   122→  durationMonths,
   123→  paymentUrl: initialPaymentUrl,
   124→  orderId: initialOrderId,
   125→  onPayNow
   126→}: PaymentInvoiceModalProps) {
   127→  const [copied, setCopied] = useState<string | null>(null)
   128→  const [paying, setPaying] = useState(false)
   129→  const [paid, setPaid] = useState(false)
   130→  const [paymentUrl, setPaymentUrl] = useState(initialPaymentUrl || '')
   131→  const [orderId, setOrderId] = useState(initialOrderId || '')
   132→
   133→  // Step 1: Select category (QRIS / E-Wallet / VA)
   134→  const [selectedCategory, setSelectedCategory] = useState<string | null>(null)
   135→  // Step 2: Select specific method within category
   136→  const [selectedMethod, setSelectedMethod] = useState<string | null>(null)
   137→  // Error state
   138→  const [payError, setPayError] = useState('')
   139→  // Manual transfer collapsed
   140→  const [showManualTransfer, setShowManualTransfer] = useState(false)
   141→
   142→  // Can pay: method selected and not already paying/paid
   143→  const canPay = selectedMethod !== null && !paying && !paid
   144→
   145→  // Get available methods for selected category
   146→  const getMethodsForCategory = (catType: string) => {
   147→    const cat = PAYMENT_CATEGORIES.find(c => c.type === catType)
   148→    return cat?.methods || []
   149→  }
   150→
   151→  // Check if a method is below minimum amount
   152→  const isBelowMin = (methodCode: string) => {
   153→    const min = METHOD_MIN_AMOUNTS[methodCode]
   154→    return min ? amount < min : false
   155→  }
   156→

   158→    try {
   159→      return new Date(dateStr).toLocaleString('id-ID', {
   160→        day: 'numeric',
   161→        month: 'long',
   162→        year: 'numeric',
   163→        hour: '2-digit',
   164→        minute: '2-digit'
   165→      })
   166→    } catch {
   167→      return dateStr
   168→    }
   169→  }
   170→
   171→  const handleCopy = (text: string, id: string) => {
   172→    navigator.clipboard.writeText(text)
   173→    setCopied(id)
   174→    setTimeout(() => setCopied(null), 2000)
   175→  }
   176→
   177→  // When user selects a category, reset the specific method
   178→  const handleCategorySelect = (catType: string) => {
   179→    if (selectedCategory === catType) {
   180→      // Deselect
   181→      setSelectedCategory(null)
   182→      setSelectedMethod(null)
   183→    } else {
   184→      setSelectedCategory(catType)
   185→      // Auto-select first valid method in category
   186→      const methods = getMethodsForCategory(catType)
   187→      const validMethod = methods.find(m => !isBelowMin(m.code))
   188→      setSelectedMethod(validMethod?.code || null)
   189→    }
   190→    setPayError('')
   191→  }
   192→
   193→  // When user selects a specific method within category
   194→  const handleMethodSelect = (methodCode: string) => {
   195→    if (isBelowMin(methodCode)) {
   196→      setPayError(`Min. pembayaran ${methodCode} adalah ${formatRupiah(METHOD_MIN_AMOUNTS[methodCode])}`)
   197→      return
   198→    }
   199→    setSelectedMethod(methodCode)
   200→    setPayError('')
   201→  }
   202→
   203→  // MAIN PAY HANDLER: Create SakuraPay order then redirect to gateway
   204→  const handlePay = async () => {
   205→    if (!selectedMethod) return
   206→    setPaying(true)
   207→    setPayError('')
   208→
   209→    try {
   210→      // If we already have a paymentUrl, open it directly
   211→      if (paymentUrl) {
   212→        window.open(paymentUrl, '_blank')
   213→        setPaying(false)
   214→        setPaid(true)
   215→        return
   216→      }
   217→
   218→      // Otherwise, create a new SakuraPay order with the selected method
   219→      const response = await fetch('/api/payment/create-order', {
   220→        method: 'POST',
   221→        headers: { 'Content-Type': 'application/json' },
   222→        body: JSON.stringify({
   223→          amount,
   224→          plan,
   225→          durationMonths,
   226→          paymentMethod: selectedMethod,
   227→        })
   228→      })
   229→
   230→      const data = await response.json()
   231→
   232→      if (!response.ok || !data.paymentUrl) {
   233→        throw new Error(data.error || data.details || 'Gagal membuat pesanan')
   234→      }
   235→
   236→      // Got paymentUrl — redirect to SakuraPay gateway
   237→      setPaymentUrl(data.paymentUrl)
   238→      setOrderId(data.orderId || '')
   239→
   240→      // Open payment gateway in new tab
   241→      window.open(data.paymentUrl, '_blank')
   242→      setPaying(false)
   243→      setPaid(true)
   244→    } catch (error: any) {
   245→      console.error('Payment error:', error)
   246→      setPaying(false)
   247→      setPayError(error.message || 'Gagal membuat pembayaran. Silakan coba lagi atau gunakan Transfer Manual.')
   248→    }
   249→  }
   250→
   251→  const getSelectedCategoryData = () => {
   252→    return PAYMENT_CATEGORIES.find(c => c.type === selectedCategory)
   253→  }
   254→
   255→  if (!isOpen) return null
   256→
   257→  return (
   258→    <AnimatePresence>
   259→      <motion.div
   260→        initial={{ opacity: 0 }}
   261→        animate={{ opacity: 1 }}
   262→        exit={{ opacity: 0 }}
   263→        className="fixed inset-0 bg-black/85 backdrop-blur-md z-[60] flex items-center justify-center p-4"
   264→        onClick={onClose}
   265→      >
   266→        <motion.div
   267→          initial={{ scale: 0.9, opacity: 0, y: 30 }}
   268→          animate={{ scale: 1, opacity: 1, y: 0 }}
   269→          exit={{ scale: 0.9, opacity: 0, y: 30 }}
   270→          transition={{ type: 'spring', damping: 25, stiffness: 300 }}
   271→          onClick={(e) => e.stopPropagation()}
   272→          className="relative w-full max-w-lg max-h-[90vh] overflow-y-auto"
   273→        >
   274→          <div className="relative bg-gradient-to-br from-[#0A0612] via-[#1A0F2E] to-[#0D0715] border border-purple-500/20 rounded-3xl overflow-hidden shadow-2xl shadow-purple-900/20">
   275→
   276→            {/* Decorative */}
   277→            <div className="absolute top-0 left-1/2 -translate-x-1/2 w-80 h-80 bg-purple-500/5 rounded-full blur-3xl" />
   278→
   279→            {/* Header */}
   280→            <div className="relative bg-gradient-to-r from-emerald-600/90 via-teal-500/90 to-emerald-600/90 backdrop-blur-xl border-b border-emerald-400/20">
   281→              <div className="absolute inset-0 bg-[url('data:image/svg+xml;base64,PHN2ZyB3aWR0aD0iNDAiIGhlaWdodD0iNDAiIHZpZXdCb3g9IjAgMCA0MCA0MCIgeG1sbnM9Imh0dHA6Ly93d3cudzMub3JnLzIwMDAvc3ZnIj48Y2lyY2xlIGN4PSIyMCIgY3k9IjIwIiByPSIxIiBmaWxsPSJyZ2JhKDI1NSwyNTUsMjU1LDAuMDUpIi8+PC9zdmc+')] opacity-50" />
   282→              <div className="relative p-6">
   283→                <div className="flex items-center justify-between">
   284→                  <div className="flex items-center gap-3">
   285→                    <div className="p-2.5 rounded-xl bg-white/15">
   286→                      <Receipt className="w-6 h-6 text-white" />
   287→                    </div>
   288→                    <div>
   289→                      <h2 className="text-xl font-bold text-white">
   290→                        {paid ? 'Pembayaran Diproses' : 'Invoice Pembayaran'}
   291→                      </h2>
   292→                      <p className="text-emerald-100/80 text-xs mt-0.5">
   293→                        {paid ? 'Selesaikan pembayaran di tab baru' : 'Pilih metode & lanjut ke pembayaran'}
   294→                      </p>
   295→                    </div>
   296→                  </div>
   297→                  <button
   298→                    onClick={onClose}
   299→                    className="p-2 rounded-full bg-white/10 hover:bg-white/20 transition-colors"

   302→                  </button>
   303→                </div>
   304→              </div>
   305→            </div>
   306→
   307→            {/* Invoice Body */}
   308→            <div className="p-6 space-y-5">
   309→              {/* Success State */}
   310→              {paid && (
   311→                <motion.div
   312→                  initial={{ opacity: 0, scale: 0.9 }}
   313→                  animate={{ opacity: 1, scale: 1 }}
   314→                  className="bg-emerald-500/10 border border-emerald-500/20 rounded-2xl p-5 flex flex-col items-center gap-3"
   315→                >
   316→                  <CheckCircle2 className="w-12 h-12 text-emerald-400" />
   317→                  <div className="text-center">
   318→                    <p className="text-lg font-bold text-emerald-300">Pesanan Dibuat!</p>
   319→                    <p className="text-sm text-white/60 mt-1">
   320→                      Selesaikan pembayaran di tab yang baru dibuka
   321→                    </p>
   322→                  </div>
   323→                  <Badge className="bg-emerald-500/20 text-emerald-300 text-xs">
   324→                    Menunggu Pembayaran
   325→                  </Badge>
   326→                </motion.div>
   327→              )}
   328→
   329→              {/* Invoice Card */}
   330→              {!paid && (
   331→                <div className="relative bg-gradient-to-br from-white/[0.04] to-white/[0.01] border border-white/10 rounded-2xl overflow-hidden">
   332→                  <div className="h-1 bg-gradient-to-r from-purple-500 via-pink-500 to-amber-500" />
   333→
   334→                  <div className="p-5 space-y-4">
   335→                    <div className="flex items-center justify-between">
   336→                      <div className="flex items-center gap-2">
   337→                        <Sparkles className="w-5 h-5 text-amber-400" />
   338→                        <span className="font-bold text-white tracking-tight">LuxTrade</span>

   341→                        INVOICE
   342→                      </Badge>
   343→                    </div>
   344→
   345→                    <div className="space-y-3">
   346→                      <div className="flex items-center justify-between text-sm">
   347→                        <span className="text-white/40">No. Invoice</span>
   348→                        <div className="flex items-center gap-2">
   349→                          <code className="text-white font-mono text-xs">{invoiceNumber}</code>
   350→                          <button onClick={() => handleCopy(invoiceNumber, 'invoice')} className="p-1 rounded hover:bg-white/10 transition-colors">
   351→                            {copied === 'invoice' ? <Check className="w-3 h-3 text-emerald-400" /> : <Copy className="w-3 h-3 text-white/40" />}
   352→                          </button>
   353→                        </div>
   354→                      </div>
   355→
   356→                      <div className="h-px bg-white/5" />

   358→                      <div className="flex items-center justify-between text-sm">

   362→
   363→                      <div className="flex items-center justify-between text-sm">
   364→                        <span className="text-white/40">Durasi</span>
   365→                        <span className="text-white">{duration}</span>
   366→                      </div>
   367→
   368→                      <div className="flex items-center justify-between text-sm">
   369→                        <span className="text-white/40">Status</span>
   370→                        <Badge className="bg-yellow-500/20 text-yellow-300 text-[10px]">

   372→                        </Badge>
   373→                      </div>
   374→
   375→                      <div className="flex items-center justify-between text-sm">
   376→                        <span className="text-white/40">Berlaku Hingga</span>
   377→                        <div className="flex items-center gap-1.5">
   378→                          <Clock className="w-3.5 h-3.5 text-amber-400/60" />

   384→                    <div className="h-px bg-gradient-to-r from-transparent via-white/10 to-transparent" />

   386→                    <div className="flex items-center justify-between">

   394→                  <div className="px-5 pb-4">

   413→                      const Icon = cat.icon

   432→                              style={{

   435→                                  cat.type === 'EWALLET' ? 'rgba(139,92,246,0.3)' :

   443→                          <div className={`relative p-2 rounded-lg transition-colors ${isSelected ? cat.color.iconBg : 'bg-white/5'}`}>

   470→                          {isSelected && (
