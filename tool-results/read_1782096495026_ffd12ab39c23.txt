   312→      console.error('❌ [TradeWizardForm] Auto-journal error:', error)

   324→
   325→    // Validate file size (5MB max)

   369→
   370→    try {
   371→      if (step === 1) {
   372→        // Validate symbol, type, lot_size, account_id
   373→        if (!formData.symbol || formData.symbol.length < 3) {
   374→          stepErrors.symbol = 'Symbol must be at least 3 characters'

   380→        if (!formData.lot_size || lot <= 0) {
   381→          stepErrors.lot_size = 'Invalid lot size'
   382→        }
   383→        if (!formData.account_id) {
   384→          stepErrors.account_id = 'Please select a trading account'
   385→        }
   386→      } else if (step === 2) {
   387→        // Validate open_price (required), close_price and profit_loss (optional for open positions)

   398→        }
   399→      }
   400→    } catch (error) {

   402→    }

   404→    return stepErrors

   417→

   446→      return

   476→        onFormChange('notes', emotionNote)
   477→      }
   478→    }
   479→
   480→    // Call onSave after updating formData
   481→    setTimeout(() => {
   482→      onSave()
   483→    }, 0)
   484→  }
   485→
   486→  const progress = (currentStep / totalSteps) * 100
   487→

   494→          <span className="text-gray-400">{Math.round(progress)}% Complete</span>
   495→        </div>
   496→        <Progress value={progress} className="h-2 bg-purple-900/30" />
   497→      </div>
   498→
   499→      {/* Step Content - Scrollable on mobile */}

   504→            key="step1"
   505→            initial={{ opacity: 0, x: 20 }}
