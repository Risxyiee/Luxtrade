    26→function generateEquityData(points: number): EquityPoint[] {
    27→  const data: EquityPoint[] = []
    28→  let equity = 10500
    29→  for (let i = 0; i < points; i++) {
    30→    const change = (Math.random() - 0.45) * 150
    31→    equity = Math.max(9500, Math.min(12000, equity + change))
    32→    data.push({ time: i, value: equity, change })
    33→  }
    34→  return data
    35→}
