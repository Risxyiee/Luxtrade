    32→// Generate realistic equity curve data
    33→function generateEquityData(points: number): EquityPoint[] {
    34→  const data: EquityPoint[] = []
    35→  let equity = 10500 // Starting equity
    36→  
    37→  for (let i = 0; i < points; i++) {
    38→    // Random walk with slight upward bias
    39→    const change = (Math.random() - 0.45) * 150
    40→    equity = Math.max(9500, Math.min(12000, equity + change))
    41→    data.push({
    42→      time: i,
    43→      value: equity,
    44→      change: change
    45→    })
    46→  }
    47→  return data
    48→}
