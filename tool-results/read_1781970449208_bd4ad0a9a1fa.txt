     1→     1→// Lazy-load Resend to avoid build-time errors when API key is missing
     2→     2→
     3→     3→export interface EmailOptions {
     4→     4→  to: string
     5→     5→  subject: string
     6→     6→  html: string
     7→     7→}
     8→     8→
     9→     9→// Get Resend client only when needed
    10→    10→async function getResendClient() {
    11→    11→  const apiKey = process.env.RESEND_API_KEY
    12→    12→  if (!apiKey) {
    13→    13→    console.warn('⚠️ RESEND_API_KEY not configured - email sending disabled')
    14→    14→    return null
    15→    15→  }
    16→    16→  // Dynamic import to avoid build-time evaluation
    17→    17→  const { Resend } = await import('resend')
    18→    18→  return new Resend(apiKey)
    19→    19→}
    20→    20→
    21→    21→export async function sendEmail({ to, subject, html }: EmailOptions) {
    22→    22→  const resend = await getResendClient()
    23→    23→
    24→    24→  if (!resend) {
    25→    25→    console.warn('⚠️ Email not sent - RESEND_API_KEY not configured')
    26→    26→    return { success: false, error: 'Email service not configured' }
    27→    27→  }
    28→    28→
    29→    29→  try {
    30→    30→    const { data, error } = await resend.emails.send({
    31→    31→      from: 'LuxTrade <noreply@luxtradee.web.id>',
    32→    32→      to,
    33→    33→      subject,
    34→    34→      html,
    35→    35→    })
    36→    36→
    37→    37→    if (error) {
    38→    38→      console.error('Resend error:', error)
    39→    39→      return { success: false, error }
    40→    40→    }
    41→    41→
    42→    42→    return { success: true, data }
    43→    43→  } catch (error) {
    44→    44→    console.error('Email send error:', error)
    45→    45→    return { success: false, error }
    46→    46→  }
    47→    47→}

    59→    59→export async function sendEmailFromTemplate({
    60→    60→  to,
    61→    61→  subject,
    62→    62→  templateId,
    63→    63→  templateParams,
    64→    64→  fallbackHtml,
    65→    65→}: {
    66→    66→  to: string
    67→    67→  subject: string
    68→    68→  templateId: string
    69→    69→  templateParams: Record<string, string>
    70→    70→  fallbackHtml: string
    71→    71→}) {
    72→    72→  const resend = await getResendClient()
    73→    73→
    74→    74→  if (!resend) {
    75→    75→    console.warn('⚠️ Email not sent - RESEND_API_KEY not configured')
    76→    76→    return { success: false, error: 'Email service not configured' }
    77→    77→  }
    78→    78→
    79→    79→  // Jika template ID belum diset, fallback ke inline HTML
    80→    80→  if (!templateId || templateId.startsWith('your_')) {
    81→    81→    console.log('📧 No template ID set, using inline HTML fallback')
    82→    82→    return sendEmail({ to, subject, html: fallbackHtml })
    83→    83→  }
    84→    84→
    85→    85→  try {
    86→    86→    const { data, error } = await resend.emails.send({
    87→    87→      from: 'LuxTrade <noreply@luxtradee.web.id>',
    88→    88→      to,
    89→    89→      subject,
    90→    90→      templateId,
    91→    91→      templateParams,
    92→    92→    })
    93→    93→
    94→    94→    if (error) {
    95→    95→      console.error('Resend template error:', error)
    96→    96→      // Fallback ke inline HTML jika template gagal
    97→    97→      console.log('📧 Template send failed, falling back to inline HTML')
    98→    98→      return sendEmail({ to, subject, html: fallbackHtml })
    99→    99→    }
   100→   100→
   101→   101→    console.log('✅ Email sent via Resend template:', templateId)
   102→   102→    return { success: true, data }
   103→   103→  } catch (error) {
   104→   104→    console.error('Email template error, falling back to inline HTML:', error)
   105→   105→    return sendEmail({ to, subject, html: fallbackHtml })
   106→   106→  }
   107→   107→}
