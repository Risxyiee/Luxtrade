     1→// Lazy-load Resend to avoid build-time errors when API key is missing
     2→
     3→export interface EmailOptions {
     4→  to: string
     5→  subject: string
     6→  html: string
     7→}
     8→
     9→// Get Resend client only when needed
    10→async function getResendClient() {
    11→  const apiKey = process.env.RESEND_API_KEY
    12→  if (!apiKey) {
    13→    console.warn('⚠️ RESEND_API_KEY not configured - email sending disabled')
    14→    return null
    15→  }
    16→  // Dynamic import to avoid build-time evaluation
    17→  const { Resend } = await import('resend')
    18→  return new Resend(apiKey)
    19→}
    20→
    21→export async function sendEmail({ to, subject, html }: EmailOptions) {
    22→  const resend = await getResendClient()
    23→
    24→  if (!resend) {
    25→    console.warn('⚠️ Email not sent - RESEND_API_KEY not configured')
    26→    return { success: false, error: 'Email service not configured' }
    27→  }
    28→
    29→  try {
    30→    const { data, error } = await resend.emails.send({
    31→      from: 'LuxTrade <noreply@luxtradee.web.id>',
    32→      to,
    33→      subject,
    34→      html,
    35→    })
    36→
    37→    if (error) {
    38→      console.error('Resend error:', error)
    39→      return { success: false, error }
    40→    }
    41→
    42→    return { success: true, data }
    43→  } catch (error) {
    44→    console.error('Email send error:', error)
    45→    return { success: false, error }
    46→  }
    47→}

    50→ * Kirim email menggunakan Resend Template dari Dashboard.
    51→ * Template variables: {{ name }}, {{ confirmationUrl }}, {{ resetUrl }}
    52→ *
    53→ * CARA SETUP:
    54→ * 1. Buka Resend Dashboard → Emails → Create Template

    56→ * 3. Copy Template ID (format: tpl_xxxxxxxx)
    57→ * 4. Set env var: RESEND_TEMPLATE_CONFIRM=tpl_xxxxxxxx dan RESEND_TEMPLATE_RESET=tpl_xxxxxxxx

    59→export async function sendEmailFromTemplate({
    60→  to,
    61→  subject,
    62→  templateId,
    63→  templateParams,
    64→  fallbackHtml,
    65→}: {
    66→  to: string
    67→  subject: string
    68→  templateId: string
    69→  templateParams: Record<string, string>
    70→  fallbackHtml: string
    71→}) {
    72→  const resend = await getResendClient()
    73→
    74→  if (!resend) {
    75→    console.warn('⚠️ Email not sent - RESEND_API_KEY not configured')
    76→    return { success: false, error: 'Email service not configured' }
    77→  }
    78→
    79→  // Jika template ID belum diset, fallback ke inline HTML
    80→  if (!templateId || templateId.startsWith('your_')) {
    81→    console.log('📧 No template ID set, using inline HTML fallback')
    82→    return sendEmail({ to, subject, html: fallbackHtml })
    83→  }
    84→
    85→  try {
    86→    const { data, error } = await resend.emails.send({
    87→      from: 'LuxTrade <noreply@luxtradee.web.id>',
    88→      to,
    89→      subject,
    90→      templateId,
    91→      templateParams,
    92→    })
    93→
    94→    if (error) {
    95→      console.error('Resend template error:', error)
    96→      // Fallback ke inline HTML jika template gagal
    97→      console.log('📧 Template send failed, falling back to inline HTML')
    98→      return sendEmail({ to, subject, html: fallbackHtml })
    99→    }
   100→
   101→    console.log('✅ Email sent via Resend template:', templateId)
   102→    return { success: true, data }
   103→  } catch (error) {
   104→    console.error('Email template error, falling back to inline HTML:', error)
   105→    return sendEmail({ to, subject, html: fallbackHtml })
   106→  }
   107→}
